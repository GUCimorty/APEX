local refreshSpeedModeRows = nil
local function setSpeedMode(mode)
  if mode~="Normal" and mode~="Carry" and mode~="Lagger" and mode~="Lagger Carry" then mode="Normal" end
  if not (autoSwitchSpeedEnabled and (carrySpeedActive or laggerCarryActive)) or mode=="Carry" or mode=="Lagger Carry" then

-- ═══════════ SPEED ENGINE ═══════════
do
  local raycastParams = RaycastParams.new()
  raycastParams.FilterType = Enum.RaycastFilterType.Exclude
  raycastParams.IgnoreWater = true
  pcall(function() raycastParams.RespectCanCollide = true end)
  local conns = {}

  local function wallSlide(hrp, char, dir, spd)
    local ok, hit = pcall(function()
      local filter = { char }
      for _, p in ipairs(P:GetPlayers()) do
        if p ~= player and p.Character then table.insert(filter, p.Character) end
      end
      raycastParams.FilterDescendantsInstances = filter
      return workspace:Raycast(hrp.Position, dir * (3 + math.min(spd or 60, 150) * 0.035), raycastParams)
    end)
    if not ok or not hit then return dir end
    if hit.Position.Y < hrp.Position.Y - 1.2 then return dir end
    local n = hit.Normal
    local horiz = math.sqrt(n.X * n.X + n.Z * n.Z)
    if horiz < 0.05 then return dir end
    local nx, nz = n.X / horiz, n.Z / horiz
    local dot = dir.X * nx + dir.Z * nz
    if dot >= 0 then return dir end
    local rx, rz = dir.X - nx * dot, dir.Z - nz * dot
    local rm = math.sqrt(rx * rx + rz * rz)
    if rm < 0.15 then return V3z end
    return V3.new(rx / rm, 0, rz / rm)
  end

  local hookedMT = nil
  local hookedHRP = nil
  local oldIdx = nil
  local oldNew = nil

  local function installHook(hrp)
    hookedHRP = hrp
    if hookedMT then return end
    pcall(function()
      local mt = getrawmetatable(game)
      if type(mt) ~= "table" then return end
      if setreadonly then pcall(setreadonly, mt, false) end
      oldIdx = rawget(mt, "__index")
      oldNew = rawget(mt, "__newindex")
      if type(oldIdx) ~= "function" or type(oldNew) ~= "function" then return end
      rawset(mt, "__index", newcclosure(function(self, key)
        if self == hookedHRP then
          if key == "AssemblyLinearVelocity" or key == "Velocity" then
            local ok, isCaller = pcall(checkcaller)
            if ok and not isCaller then return cachedVelocity end
          end
        end
        return oldIdx(self, key)
      end))
      rawset(mt, "__newindex", newcclosure(function(self, key, value)
        if self == hookedHRP then
          if key == "AssemblyLinearVelocity" or key == "Velocity" then
            local ok, isCaller = pcall(checkcaller)
            if ok and not isCaller then cachedVelocity = value; return end
          end
        end
        return oldNew(self, key, value)
      end))
      if setreadonly then pcall(setreadonly, mt, true) end
      hookedMT = mt
    end)
  end

  fn56 = function()
    for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
    conns = {}
    local char = player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    pcall(installHook, hrp)

    local c = RS.PreSimulation:Connect(safe(function()

      local hum = tbl22.humanoid and tbl22.humanoid.Parent and tbl22.humanoid or char:FindFirstChildOfClass("Humanoid")
      if not (hum and hum.Health > 0 and hrp and hrp.Parent) then return end
      if hrp.Position.Y < -20 then return end
      local spd = getCurrentSpeedValue()
      if spd >= 15.5 and spd <= 16.5 then cachedVelocity = V3z; return end
      local md = hum.MoveDirection
      local mag = md.Magnitude
      if mag > 0.05 then
        local y = hrp.AssemblyLinearVelocity.Y
        if y ~= y then y = 0 end
        local cy = clamp(y, -80, 80)
        local dir = wallSlide(hrp, char, V3(md.X / mag, 0, md.Z / mag), spd)
        if dir.Magnitude < 0.05 then
          cachedVelocity = V3(0, cy, 0)
          hrp.AssemblyLinearVelocity = V3(0, cy, 0)
        else
          cachedVelocity = V3(dir.X * 20, cy, dir.Z * 16)
          hrp.AssemblyLinearVelocity = V3(dir.X * spd, cy, dir.Z * spd)
        end
      elseif cachedVelocity ~= V3z then
        cachedVelocity = V3z
      end
    end))
    table.insert(conns, c)
  end

  pcall(fn56)
  player.CharacterAdded:Connect(function() task.wait(0.1); pcall(fn56) end)

  pcall(function()
    workspace:GetPropertyChangedSignal("Gravity"):Connect(function()
      if workspace.Gravity > 300 then workspace.Gravity = 196.2 end
    end)
    if workspace.Gravity > 300 then workspace.Gravity = 196.2 end
  end)

  fn54 = function(hrp, hum, char, dir, spd, dt)
    if not hrp or not char then return end

    if dir.Magnitude > 0.1 then

      local unit = flat.Magnitude > 0.05 and flat.Unit or V3z
      local d2 = unit.Magnitude > 0.1 and wallSlide(hrp, char, unit, spd) or V3z

        cachedVelocity = V3(0, y, 0)
        hrp.AssemblyLinearVelocity = V3(0, y, 0)
      else
        cachedVelocity = V3(d2.X * 16, y, d2.Z * 16)
        hrp.AssemblyLinearVelocity = V3(d2.X * spd, y, d2.Z * spd)
      end
    else
      cachedVelocity = V3(0, y, 0)
    end
  end
end

                if string.find(t, lname, 1, true) or string.find(t, ldisp, 1, true) then mine = true; break end
              end
            end
          end
        end

          local sp = child:FindFirstChild("Spawn") or child:FindFirstChild("Plot") or child:FindFirstChild("Base")

          Steal.StealState.active = false; Steal.StealState.phase = "idle"
          Steal.StealState.lastResultTime = tick()

        stealSetProgress(math.min(n / Steal.StealDuration, 1))

local FpsBoostData = {State={}, Active=false, Connection=nil}
function FpsBoost(Toggle)
  pcall(function()
    local function saveState()
      if FpsBoostData.State.Saved then return end
      FpsBoostData.State.Lighting={GlobalShadows=L.GlobalShadows, FogEnd=L.FogEnd, FogStart=L.FogStart,
        EnvironmentDiffuseScale=L.EnvironmentDiffuseScale, EnvironmentSpecularScale=L.EnvironmentSpecularScale}
      FpsBoostData.State.Effects={}
      for _,v in pairs(L:GetChildren()) do
        if v:IsA("BloomEffect") or v:IsA("BlurEffect") or v:IsA("ColorCorrectionEffect")
          or v:IsA("DepthOfFieldEffect") or v:IsA("Atmosphere") then
          FpsBoostData.State.Effects[#FpsBoostData.State.Effects+1]=v:Clone()
        end
      end
      FpsBoostData.State.Saved=true
    end
    local function apply(obj)
      pcall(function()
        local model = obj:FindFirstAncestorOfClass("Model")
        local isPlayer = model and P:GetPlayerFromCharacter(model)~=nil
        if obj:IsA("Animator") then
          local am = obj:FindFirstAncestorOfClass("Model")
          if not (am and P:GetPlayerFromCharacter(am)) then
            for _,t in pairs(obj:GetPlayingAnimationTracks()) do pcall(function() t:Stop(0) end) end
          end
        end
        if not isPlayer then
          if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") or obj:IsA("Smoke")
            or obj:IsA("Fire") or obj:IsA("Sparkles") or obj:IsA("Highlight") then obj.Enabled=false end
          pcall(function() if obj:IsA("Explosion") then obj:Destroy() end end)
        end
        if obj:IsA("BasePart") then obj.Material=Enum.Material.Plastic; obj.Reflectance=0; obj.CastShadow=false end
      end)
    end
    if Toggle==nil then Toggle = not FpsBoostData.Active end
    if Toggle and not FpsBoostData.Active then
      saveState()
      L.GlobalShadows=false; L.FogEnd=1e6; L.FogStart=0; L.EnvironmentDiffuseScale=0; L.EnvironmentSpecularScale=0
      for _,v in pairs(L:GetChildren()) do
        if v:IsA("BloomEffect") or v:IsA("BlurEffect") or v:IsA("ColorCorrectionEffect")
          or v:IsA("DepthOfFieldEffect") or v:IsA("Atmosphere") then v:Destroy()
        elseif v:IsA("SunRaysEffect") then v.Enabled=false end
      end
      local d=workspace:GetDescendants(); local bs=500; local tot=#d
      for i=1,tot,bs do local lim=math.min(i+bs-1,tot); for idx=i,lim do apply(d[idx]) end; if lim<tot then task.wait() end end
      if FpsBoostData.Connection then FpsBoostData.Connection:Disconnect() end
      FpsBoostData.Connection = workspace.DescendantAdded:Connect(safe(function(o) if FpsBoostData.Active then apply(o) end end))
      FpsBoostData.Active=true
    elseif not Toggle and FpsBoostData.Active then
      if FpsBoostData.Connection then FpsBoostData.Connection:Disconnect(); FpsBoostData.Connection=nil end
      if FpsBoostData.State.Lighting then
        L.GlobalShadows=FpsBoostData.State.Lighting.GlobalShadows
        L.FogEnd=FpsBoostData.State.Lighting.FogEnd
        L.FogStart=FpsBoostData.State.Lighting.FogStart
        L.EnvironmentDiffuseScale=FpsBoostData.State.Lighting.EnvironmentDiffuseScale
        L.EnvironmentSpecularScale=FpsBoostData.State.Lighting.EnvironmentSpecularScale
      end
      for _,e in pairs(FpsBoostData.State.Effects or {}) do pcall(function() e:Clone().Parent=L end) end
      FpsBoostData.State={}; FpsBoostData.Active=false
    end
  end)
end

local function getClosestAim()
  local root=player.Character and player.Character:FindFirstChild("HumanoidRootPart")
  if not root then return nil,math.huge end
  local closest,minDist=nil,math.huge
  for _,plr in ipairs(P:GetPlayers()) do
    if plr~=player and plr.Character then
      local tr=plr.Character:FindFirstChild("HumanoidRootPart")
      local hum=plr.Character:FindFirstChildOfClass("Humanoid")
      if tr and hum and hum.Health>0 then
        local d=(tr.Position-root.Position).Magnitude
        if d<minDist then minDist=d; closest=plr end
      end
    end
  end
  return closest,minDist
end

      if BatAimbot.align then BatAimbot.align.Enabled = false end
      hum2.AutoRotate = true
      BatAimbot.target = nil
      return
    end
    local tr = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
    if not tr then return end

      local vy = (tr.Position.Y + 3.7 - root.Position.Y) * 19.5
      if hum2.FloorMaterial ~= Enum.Material.Air then vy = math.max(vy, 13) end

        if bat then pcall(function() hum2:EquipTool(bat) end) end
      end
      if dist and dist <= 10 and not BatAimbot.hitCooling then
        BatAimbot.hitCooling = true

          local bat = character2:FindFirstChildOfClass("Tool") or findBat()
          if bat and bat.Parent == character2 then
            bat:Activate()
            local re = bat:FindFirstChildWhichIsA("RemoteEvent")
            if re then re:FireServer() end
          end
        end)

      end

      hum2.AutoRotate = false
      if not character2:FindFirstChildOfClass("Tool") then
        local bat = findBat()
        if bat then pcall(function() hum2:EquipTool(bat) end) end

      local unit = flat.Magnitude > 0 and flat.Unit or V3z
      local ySpd = (tPos.Y + 3.7 - myPos.Y) * 19.5 + tVel.Y * 0.8

        local rx, ry, rz = (root.CFrame:Inverse() * cf):ToEulerAnglesXYZ()
        root.AssemblyAngularVelocity = root.CFrame:VectorToWorldSpace(V3(
          clamp(rx, -1.6, 1.6) * 42, clamp(ry, -1.6, 2.5) * 42, clamp(rz, -1.6, 2.5) * 42))

    lbl.TextColor3 = C3(255,255,255); lbl.Visible = true
    task.spawn(safe(function()
      while true do
        if state.active and state.gen==myGen then
          local left = ragCountdownDur - (tick()-t0)
          if left > 0 then
            local l = getLabel()
            if l and l.Parent then
              l.Text = string.format("%.1fs", left)
              if left <= 0.8 then l.TextColor3 = C3(255,80,80)
              elseif left <= 1.5 then l.TextColor3 = C3(255,200,80)
              else l.TextColor3 = C3(255,255,255) end
            end
            task.wait(0.03)
          else break end
        else break end
      end
      if state.gen==myGen then
        local l = getLabel(); if l and l.Parent then l.Visible=false end
        state.active = false
      end
    end))
  end
  local function bind(char)
    for _,c in ipairs(state.conns) do pcall(function() c:Disconnect() end) end
    state.conns = {}
    if not char then return end
    local hum = char:WaitForChild("Humanoid", 4)
    if hum then
      table.insert(state.conns, hum.StateChanged:Connect(safe(function(_,n)
        if not ragTimerEnabled or state.active then return end
        if n==Enum.HumanoidStateType.Physics or n==Enum.HumanoidStateType.Ragdoll
          or n==Enum.HumanoidStateType.FallingDown or n==Enum.HumanoidStateType.PlatformStanding then start() end
      end)))
    end
    table.insert(state.conns, char.ChildAdded:Connect(safe(function(c)
      if not ragTimerEnabled or state.active then return end
      local n = c.Name:lower()
      if n:find("ragdoll") or n:find("knock") or n:find("stun") then start() end
    end)))
    table.insert(state.conns, RS.Heartbeat:Connect(safe(function()
      if not char.Parent or not ragTimerEnabled then return end
      if not state.active and isRagdolled(char) then start() end
    end)))
  end
  function ragTimerSetEnabled(on)
    ragTimerEnabled = on
    if on then
      state.on = true
      if player.Character then pcall(bind, player.Character) end
      if not state.charConn then
        state.charConn = player.CharacterAdded:Connect(function(c)
          if ragTimerEnabled then task.wait(0.15); pcall(bind, c) end
        end)
      end
    else
      state.on = false
      if state.charConn then state.charConn:Disconnect(); state.charConn=nil end
      for _,c in ipairs(state.conns) do pcall(function() c:Disconnect() end) end
      state.conns = {}
      state.active = false; state.gen = state.gen+1
      local l = getLabel(); if l and l.Parent then l.Visible=false end
    end
  end
end

    if match(KB.LaggerToggle) then
      if tick()-_lastLaggerToggle>=0.3 then
        _lastLaggerToggle=tick()
        toggleLaggerMode()
      end
      return
    end
    if match(KB.DropBrainrot) then runDrop();return end
    if match(KB.TPDown) then runTPDown(true);return end
    if match(KB.BatAimbot) then if batAimbotEnabled then stopBatAimbot() else startBatAimbot() end;task.spawn(function() pcall(saveConfig) end);return end

    if match(KB.AutoLeft) then if autoLeftEnabled then stopAutoLeft() else startAutoLeft() end;task.spawn(function() pcall(saveConfig) end);return end

    if match(KB.WalkLeft) then if fn56_walk then pcall(fn56_walk) end;return end
